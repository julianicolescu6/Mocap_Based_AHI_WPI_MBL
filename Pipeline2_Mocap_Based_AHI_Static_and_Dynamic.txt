! Run this pipeline only after running pipeline 1 on the static trial
! Make sure the model has been assigned to all desired trials (including static)
! This pipeline performs truncated foot length, dorsum height, and AHI calculations and exports results to a .txt file

! Right Foot

Compute_Model_Based_Data
/RESULT_NAME=Right_Proximal_MT2
/FUNCTION=TARGET_PATH
/SEGMENT=R_MTII_prox
/REFERENCE_SEGMENT=RFT
/RESOLUTION_COORDINATE_SYSTEM=RFT
! /USE_CARDAN_SEQUENCE=FALSE
! /NORMALIZATION=FALSE
! /NORMALIZATION_METHOD=
! /NORMALIZATION_METRIC=
! /NEGATEX=FALSE
! /NEGATEY=FALSE
! /NEGATEZ=FALSE
! /AXIS1=X
! /AXIS2=Y
! /AXIS3=Z
! /TREADMILL_DATA=FALSE
! /TREADMILL_DIRECTION=UNIT_VECTOR(0,1,0)
! /TREADMILL_SPEED=0.0
;

Compute_Model_Based_Data
/RESULT_NAME=Right_Distal_MT1
/FUNCTION=TARGET_PATH
/SEGMENT=R_MTI_dist
/REFERENCE_SEGMENT=RFT
/RESOLUTION_COORDINATE_SYSTEM=RFT
! /USE_CARDAN_SEQUENCE=FALSE
! /NORMALIZATION=FALSE
! /NORMALIZATION_METHOD=
! /NORMALIZATION_METRIC=
! /NEGATEX=FALSE
! /NEGATEY=FALSE
! /NEGATEZ=FALSE
! /AXIS1=X
! /AXIS2=Y
! /AXIS3=Z
! /TREADMILL_DATA=FALSE
! /TREADMILL_DIRECTION=UNIT_VECTOR(0,1,0)
! /TREADMILL_SPEED=0.0
;

Metric_Explicit
!/RESULT_METRIC_FOLDER=PROCESSED
/RESULT_METRIC_NAME=Right_Truncated_Foot_Length
/METRIC_VALUE=LINK_MODEL_BASED::ORIGINAL::Right_Distal_MT1::Y-0.00475
;

Metric_Explicit
!/RESULT_METRIC_FOLDER=PROCESSED
/RESULT_METRIC_NAME=Right_Dorsum_Height
/METRIC_VALUE=(ADD(LINK_MODEL_BASED::ORIGINAL::Right_Proximal_MT2::Z,GLOBAL::METRIC::PROCESSED::Static_Right_Segment_Height::Z))-0.00475
;

Metric_Explicit
!/RESULT_METRIC_FOLDER=PROCESSED
/RESULT_METRIC_NAME=Right_AHI
/METRIC_VALUE= METRIC::PROCESSED::Right_Dorsum_Height/METRIC::PROCESSED::Right_Truncated_Foot_Length
;

Export_Data_To_Ascii_File
/SIGNAL_TYPES=METRIC
/SIGNAL_FOLDER=PROCESSED
/SIGNAL_NAMES=Right_Dorsum_Height+Right_Truncated_Foot_Length+Right_AHI
! /FILE_NAME=
! /SIGNAL_COMPONENTS=
!/COMPONENT_SEQUENCE=
!/SIGNAL_PRECISION=
! /EVENT_SEQUENCE=
! /EXCLUDE_EVENTS=
! /USE_POINT_RATE=FALSE
! /NORMALIZE_DATA=FALSE
! /NORMALIZE_POINTS=101
! /EXPORT_MEAN_AND_STD_DEV=FALSE
! /EXPORT_MEAN_AND_STD_DEV_FOR_METRIC=FALSE
! /USE_P2D_FORMAT=FALSE
! /USE_XML_FORMAT=FALSE
! /USE_JSON_FORMAT=FALSE
! /USE_SHORT_FILENAME=FALSE
! /INCLUDE_TAGS_FOR_XML=FALSE
! /EXPORT_EMPTY_SIGNALS=FALSE
! /EXPORT_WITHOUT_HEADER=FALSE
! /EXPORT_NAN=FALSE
! /USE_SCIENTIFIC_NOTATION=FALSE
;
